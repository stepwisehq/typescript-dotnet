## Early demonstration candidate

The following is a dependency-driving draft, not yet an actual-Go passing fixture:

```ts
interface Sale { team: string; amount: number; active: boolean }
type Groups<T> = Record<string, T[]>;
function group<T>(values: readonly T[], key: (value: T) => string): Groups<T> {
  const result: Groups<T> = {};
  for (const value of values) {
    const name = key(value);
    const bucket = result[name];
    if (bucket) bucket.push(value);
    else result[name] = [value];
  }
  return result;
}
function label(value: number): string;
function label(value: string): string;
function label(value: number | string): string { return "" + value; }
function copySummary(...items: [string, number]): [string, number] { return items; }
const sales: Sale[] = [
  { team: "north", amount: 12, active: true },
  { team: "south", amount: 7, active: false }
];
const active = sales.filter(row => row.active);
const groups = group(active, row => row.team);
const amounts = active.map(row => row.amount);
const names = amounts.map(label);
const total = amounts.reduce((sum, amount) => sum + amount, 0);
const summary: [string, number] = [label(total), total];
const copied = copySummary(...summary);
```
